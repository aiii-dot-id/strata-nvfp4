cmake_minimum_required(VERSION 3.24)
project(strata_ring_reference LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(STRATA_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_library(ring_reference STATIC
  ${STRATA_ROOT}/src/ring/attention.cpp
  ${STRATA_ROOT}/src/ring/routing.cpp
  ${STRATA_ROOT}/src/ring/expert_cache.cpp
  ${STRATA_ROOT}/src/ring/tensor_store.cpp)
target_include_directories(ring_reference PUBLIC ${STRATA_ROOT}/include)
enable_testing()
add_executable(terminal_adapter_test terminal_adapter_test.cpp)
target_include_directories(terminal_adapter_test PRIVATE ${STRATA_ROOT}/include)
add_test(NAME terminal_adapter COMMAND terminal_adapter_test)
foreach(unit attention routing expert_cache tensor_store numerics generation continuation)
  add_executable(ring_${unit}_test ${STRATA_ROOT}/tests/ring_${unit}_test.cpp)
  target_link_libraries(ring_${unit}_test PRIVATE ring_reference)
  target_compile_options(ring_${unit}_test PRIVATE -UNDEBUG)
  add_test(NAME ring_${unit} COMMAND ring_${unit}_test)
endforeach()
option(RING_ENABLE_CUDA "Build opt-in BF16 Ring reference executable" OFF)
option(RING_ENABLE_TORCH_ROUTER "Build optional ATen CUDA router backend" OFF)
if(RING_ENABLE_CUDA)
  if(RING_ENABLE_TORCH_ROUTER)
    find_package(Torch REQUIRED)
    separate_arguments(STRATA_TORCH_CXX_FLAGS NATIVE_COMMAND "${TORCH_CXX_FLAGS}")
    add_library(torch_router_backend STATIC torch_router/router.cpp torch_router/rms.cpp
      torch_router/attention.cpp torch_router/pointwise.cpp)
    target_compile_options(ring_reference PUBLIC ${STRATA_TORCH_CXX_FLAGS})
    target_compile_features(torch_router_backend PUBLIC cxx_std_20)
    target_include_directories(torch_router_backend PUBLIC ${STRATA_ROOT}/include ${CMAKE_CURRENT_SOURCE_DIR}/torch_router ${TORCH_INCLUDE_DIRS})
    target_link_libraries(torch_router_backend PUBLIC ${TORCH_LIBRARIES})
    target_compile_definitions(torch_router_backend PUBLIC STRATA_TORCH_ROUTER=1)
    target_compile_options(torch_router_backend PRIVATE ${STRATA_TORCH_CXX_FLAGS})
    add_executable(ring_torch_router_test torch_router/router_test.cpp)
    target_compile_features(ring_torch_router_test PRIVATE cxx_std_20)
    target_link_libraries(ring_torch_router_test PRIVATE torch_router_backend)
    target_compile_options(ring_torch_router_test PRIVATE ${STRATA_TORCH_CXX_FLAGS} -UNDEBUG)
    add_test(NAME ring_torch_router COMMAND ring_torch_router_test)
  endif()
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  set(CMAKE_CUDA_STANDARD 17)
  set(CMAKE_CUDA_STANDARD_REQUIRED ON)
  add_executable(strata_ring_bf16
    ${STRATA_ROOT}/src/ring/engine.cu
    ${STRATA_ROOT}/src/kernels/cuda/native_bf16.cu)
  set_target_properties(strata_ring_bf16 PROPERTIES CUDA_ARCHITECTURES 89)
  target_compile_options(strata_ring_bf16 PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-Xcompiler=-ffunction-sections>)
  target_link_options(strata_ring_bf16 PRIVATE -Wl,--gc-sections)
  target_link_libraries(strata_ring_bf16 PRIVATE ring_reference CUDA::cublas)
  if(RING_ENABLE_TORCH_ROUTER)
    target_link_libraries(strata_ring_bf16 PRIVATE torch_router_backend)
    target_compile_definitions(strata_ring_bf16 PRIVATE STRATA_TORCH_ROUTER=1)
    target_include_directories(strata_ring_bf16 PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/torch_router)
    target_compile_options(strata_ring_bf16 PRIVATE ${STRATA_TORCH_CXX_FLAGS})
  endif()
endif()
